% Beauty Trap
% Perform the most seductive dance

enum LEGS = {spin, leap, waltz, curtsey, prep, stand};
enum ARMS = {beckon, out, up, wrapped, neutral};
enum FACE = {smile, wink, batt, think, glow, blank};

int: maxlen;
set of int: STEP = 1..maxlen;
array[LEGS] of int: maxlegs;
array[ARMS] of int: maxarms;
array[FACE] of int: maxface;
constraint assert(maxlegs[stand] >= maxlen, "maxlegs[stand] smaller than maxlen");
constraint assert(maxarms[neutral] >= maxlen, "maxarms[neutral] smaller than maxlen");
constraint assert(maxface[blank] >= maxlen, "maxface[blank] smaller than maxlen");
array[LEGS,ARMS] of int: dance_value;
array[ARMS,FACE] of int: entice_value;
constraint assert(dance_value[stand,neutral] = 0, "incorrect dance_value array");
constraint assert(entice_value[neutral,blank] = 0, "incorrect entice_value array");
int: boredom; % how bored each step make the viewer

var STEP: len;
array[STEP] of var LEGS: legs;
array[STEP] of var ARMS: arms;
array[STEP] of var FACE: face;

include "regular.mzn";

constraint forall(i in LEGS)(sum(j in 1..maxlen)(legs[j] = i) <= maxlegs[i]);
constraint forall(i in ARMS)(sum(j in 1..maxlen)(arms[j] = i) <= maxarms[i]);
constraint forall(i in FACE)(sum(j in 1..maxlen)(face[j] = i) <= maxface[i]);

constraint regular(legs, 7, LEGS, dl, 1, {7});
array[1..7, LEGS] of 0..7: dl = 
[| 2, 0, 4, 5, 6, 7
 | 0, 0, 0, 5, 6, 7
 | 2, 0, 4, 0, 0, 7
 | 2, 0, 4, 0, 6, 7
 | 2, 0, 4, 5, 0, 7
 | 0, 3, 0, 0, 0, 0
 | 2, 0, 4, 5, 6, 7
 |];
 
constraint forall(i in 1..len-1)((legs[i] = waltz -> 
forall(j in i+1..len)(legs[j] = curtsey -> exists(x in i+1..j-1)(legs[x] = stand))));
 
constraint forall(i in 1..maxlen-3)((legs[i] = waltz /\ legs[i+1] = waltz /\ legs[i+2] = waltz) -> legs[i+3] != waltz);

constraint regular(arms, 6, ARMS, da, 1, {6});
array[1..6, ARMS] of 0..6: da = 
[| 0, 3, 4, 5, 6
 | 0, 3, 4, 5, 6
 | 2, 0, 4, 0, 6
 | 2, 3, 0, 0, 6
 | 0, 3, 4, 0, 6
 | 0, 3, 4, 5, 6
 |];
 
 
constraint regular(face, 7, FACE, df, 1, {7});
array[1..7, FACE] of 0..7: df = 
[| 2, 3, 4, 5, 6, 7
 | 2, 3, 4, 0, 6, 7
 | 2, 3, 4, 5, 6, 7
 | 2, 3, 4, 0, 6, 7
 | 2, 3, 4, 5, 6, 7
 | 2, 3, 4, 5, 6, 7
 | 2, 3, 4, 5, 6, 7
 |];
 
constraint forall(i in 1..maxlen-2)((face[i] != blank /\ face[i] = face[i+1]) -> face[i + 2] != face[i]);

constraint forall(i in 1..maxlen)(dance_value[legs[i], arms[i]] != -1 /\ entice_value[arms[i], face[i]] != -1);
 
constraint len = length(legs) - 1;

constraint len <= maxlen;

var int: obj;
 
constraint obj = sum([dance_value[legs[i], arms[i]] | i in 1..len]) + sum([entice_value[arms[i], face[i]] | i in 1..len]) - (boredom * len);

solve maximize obj;
